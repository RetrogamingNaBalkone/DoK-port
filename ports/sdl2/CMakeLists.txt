cmake_minimum_required(VERSION 3.20)
project(dok_source_port VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DOK_BUILD_APP "Build the SDL2 application" ON)
add_library(dok_core src/resources.cpp src/level.cpp src/world.cpp src/events.cpp src/camera.cpp src/raster.cpp src/geometry.cpp src/scene.cpp src/movement.cpp src/interactions.cpp src/items.cpp src/weapons.cpp src/projectiles.cpp src/monsters.cpp)
target_include_directories(dok_core PUBLIC include)
target_sources(dok_core PRIVATE src/flow.cpp)
target_sources(dok_core PRIVATE src/hud.cpp)
target_sources(dok_core PRIVATE src/hud_desktop.cpp)
target_sources(dok_core PRIVATE src/text.cpp)
target_sources(dok_core PRIVATE src/audio.cpp)
target_sources(dok_core PRIVATE src/png.cpp)
target_sources(dok_core PRIVATE src/gpu_geometry.cpp)
if(MSVC)
  target_compile_options(dok_core PRIVATE /W4)
else()
  target_compile_options(dok_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(dok_inspect src/inspect.cpp src/diagnostics.cpp)
target_link_libraries(dok_inspect PRIVATE dok_core)
include(CTest)

if(DOK_BUILD_APP)
  find_package(SDL2 CONFIG REQUIRED)
  add_library(dok_sdl_input src/sdl_input.cpp)
  target_sources(dok_sdl_input PRIVATE src/sdl_audio.cpp)
  target_sources(dok_sdl_input PRIVATE src/gl_renderer.cpp)
  target_link_libraries(dok_sdl_input PUBLIC dok_core SDL2::SDL2)
  if(BUILD_TESTING)
    add_executable(dok_hud_desktop_probe tests/hud_desktop_probe.cpp)
    target_compile_definitions(dok_hud_desktop_probe PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(dok_hud_desktop_probe PRIVATE dok_sdl_input)
    add_executable(dok_gl_probe tests/gl_probe.cpp)
    target_compile_definitions(dok_gl_probe PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(dok_gl_probe PRIVATE dok_sdl_input)
    add_executable(dok_input_tests tests/input_tests.cpp)
    target_compile_definitions(dok_input_tests PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(dok_input_tests PRIVATE dok_sdl_input)
    add_test(NAME input COMMAND dok_input_tests)
  endif()
  add_executable(dok_port src/main.cpp)
  target_compile_definitions(dok_port PRIVATE SDL_MAIN_HANDLED)
  target_link_libraries(dok_port PRIVATE dok_core dok_sdl_input SDL2::SDL2)
  if(WIN32)
    add_custom_command(TARGET dok_port POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:dok_port>)
  endif()
endif()
